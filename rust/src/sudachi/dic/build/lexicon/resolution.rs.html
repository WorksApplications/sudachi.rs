27impl LexiconReader {
28    pub(crate) fn resolve_entries<R: WordRefResolver>(
29        &mut self,
30        resolver: &R,
31        user: bool,
32    ) -> Result<usize, (String, usize)> {
33        let mut total = 0;
34        let mut resolved_entries = Vec::with_capacity(self.parsed_entries.len());
35        let mut phantom_resolved: Vec<ResolvedLexiconEntry> = Vec::new();
36        let mut phantom_headwords: HashMap<String, DicWordRef> = HashMap::new();
37        let mut next_phantom_offset =
38            (self.next_entry_id() as usize) << WordInfos::WORD_ID_ALIGNMENT_BITS;
39        for (line, entry) in self.parsed_entries.iter().cloned().enumerate() {
40            let (resolved, resolved_count, phantom_headword) = self
41                .resolve_entry(
42                    entry,
43                    resolver,
44                    &phantom_headwords,
45                    next_phantom_offset,
46                    user,
47                )
48                .map_err(|split_info| (split_info, line))?;
49            total += resolved_count;
50            if let Some(headword) = phantom_headword {
51                phantom_resolved.push(ResolvedLexiconEntry::make_phantom(&resolved, headword));
52                let phantom_ref = DicWordRef::new(
53                    !user,
54                    (next_phantom_offset >> WordInfos::WORD_ID_ALIGNMENT_BITS) as u32,
55                );
56                phantom_headwords.insert(
57                    phantom_resolved.last().unwrap().headword().to_owned(),
58                    phantom_ref,
59                );
60                next_phantom_offset += phantom_resolved.last().unwrap().expected_entry_size();
61            }
62            resolved_entries.push(resolved);
63        }
64        resolved_entries.extend(phantom_resolved);
65        self.resolved_entries = resolved_entries;
66        Ok(total)
67    }

69    pub(crate) fn row_word_ids(&self, dic_id: u8) -> Vec<WordId> {
70        let mut result = Vec::with_capacity(self.parsed_entries.len());
71        let mut offset = Self::ENTRY_INITIAL_OFFSET;
72        for e in &self.parsed_entries {
73            let entry_id = (offset >> WordInfos::WORD_ID_ALIGNMENT_BITS) as u32;
74            result.push(WordId::new(dic_id, entry_id));
75            offset += e.expected_entry_size();
76        }
77        result
78    }
79
80    pub(crate) fn row_word_refs(&self, user: bool) -> Vec<DicWordRef> {
81        let mut result = Vec::with_capacity(self.parsed_entries.len());
82        let mut offset = Self::ENTRY_INITIAL_OFFSET;
83        for e in &self.parsed_entries {
84            let entry_id = (offset >> WordInfos::WORD_ID_ALIGNMENT_BITS) as u32;
85            result.push(DicWordRef::new(!user, entry_id));
86            offset += e.expected_entry_size();
87        }
88        result
89    }

91    fn resolve_entry<R: WordRefResolver>(
92        &self,
93        entry: ParsedLexiconEntry,
94        resolver: &R,
95        phantom_headwords: &HashMap<String, DicWordRef>,
96        next_phantom_offset: usize,
97        user: bool,
98    ) -> Result<(ResolvedLexiconEntry, usize, Option<String>), String> {
99        let mut total = 0;
100        let (norm_form, phantom_headword) = self.resolve_norm_form(
101            &entry,
102            resolver,
103            phantom_headwords,
104            next_phantom_offset,
105            user,
106            &mut total,
107        )?;
108        let dic_form = self
109            .resolve_dic_form_ref(&entry.dic_form, resolver, &mut total)
110            .map_err(|r| self.format_word_ref(&r))?;
111        let splits_a = self
112            .resolve_word_refs(&entry.splits_a, resolver, &mut total)
113            .map_err(|r| self.format_word_ref(&r))?;
114        let splits_b = self
115            .resolve_word_refs(&entry.splits_b, resolver, &mut total)
116            .map_err(|r| self.format_word_ref(&r))?;
117        let splits_c = self
118            .resolve_word_refs(&entry.splits_c, resolver, &mut total)
119            .map_err(|r| self.format_word_ref(&r))?;
120        let word_structure = self
121            .resolve_word_refs(&entry.word_structure, resolver, &mut total)
122            .map_err(|r| self.format_word_ref(&r))?;
123
124        let resolved = ResolvedLexiconEntry {
125            left_id: entry.left_id,
126            right_id: entry.right_id,
127            cost: entry.cost,
128            index_form: entry.index_form,
129            headword: entry.headword,
130            reference_id: entry.reference_id,
131            dic_form,
132            norm_form,
133            pos: entry.pos,
134            splits_a,
135            splits_b,
136            splits_c,
137            reading: entry.reading,
138            splitting: entry.splitting,
139            word_structure,
140            synonym_groups: entry.synonym_groups,
141            user_data: entry.user_data,
142        };
143
144        Ok((resolved, total, phantom_headword))
145    }

147    fn resolve_norm_form<R: WordRefResolver>(
148        &self,
149        entry: &ParsedLexiconEntry,
150        resolver: &R,
151        phantom_headwords: &HashMap<String, DicWordRef>,
152        next_phantom_offset: usize,
153        user: bool,
154        total: &mut usize,
155    ) -> Result<(ResolvedWordRef, Option<String>), String> {
156        match &entry.norm_form {
157            WordRef::SelfRef => Ok((ResolvedWordRef::SelfRef, None)),
158            WordRef::Headword(headword) => {
159                *total += 1;
160                if let Some(wref) = resolver
161                    .resolve_by_headword(headword)
162                    .or_else(|| phantom_headwords.get(headword).copied())
163                {
164                    return Ok((ResolvedWordRef::Ref(wref), None));
165                }
166
167                // If the given string does not found in the resolver/previous entries,
168                // add phantom entry with that headword.
169                let phantom_ref = DicWordRef::new(
170                    !user,
171                    (next_phantom_offset >> WordInfos::WORD_ID_ALIGNMENT_BITS) as u32,
172                );
173                Ok((ResolvedWordRef::Ref(phantom_ref), Some(headword.to_owned())))
174            }
175            word_ref => {
176                let wref = resolver
177                    .resolve(word_ref)
178                    .ok_or_else(|| self.format_word_ref(word_ref))?;
179                *total += 1;
180                Ok((ResolvedWordRef::Ref(wref), None))
181            }
182        }
183    }
184
185    fn resolve_dic_form_ref<R: WordRefResolver>(
186        &self,
187        word_ref: &WordRef,
188        resolver: &R,
189        total: &mut usize,
190    ) -> Result<ResolvedWordRef, WordRef> {
191        match word_ref {
192            WordRef::SelfRef => Ok(ResolvedWordRef::SelfRef),
193            other => {
194                let wref = resolver.resolve(other).ok_or_else(|| other.clone())?;
195                *total += 1;
196                Ok(ResolvedWordRef::Ref(wref))
197            }
198        }
199    }
200
201    fn resolve_word_refs<R: WordRefResolver>(
202        &self,
203        word_refs: &[WordRef],
204        resolver: &R,
205        total: &mut usize,
206    ) -> Result<Vec<DicWordRef>, WordRef> {
207        let mut out = Vec::with_capacity(word_refs.len());
208        for word_ref in word_refs {
209            let wref = resolver.resolve(word_ref).ok_or_else(|| word_ref.clone())?;
210            match word_ref {
211                WordRef::SelfRef => {}
212                _ => *total += 1,
213            }
214            out.push(wref);
215        }
216        Ok(out)
217    }
218
219    fn format_word_ref(&self, word_ref: &WordRef) -> String {
220        match word_ref {
221            WordRef::SelfRef => "<self>".to_owned(),
222            WordRef::LineRef(id) => id.as_raw().to_string(),
223            WordRef::Headword(h) => h.clone(),
224            WordRef::EntryKey {
225                headword,
226                pos,
227                reading,
228                reference_id,
229            } => format!(
230                "{},{:?},{}{}",
231                headword,
232                self.pos_obj(*pos).unwrap(),
233                reading,
234                reference_id
235                    .as_ref()
236                    .map(|rid| format!(",{}", rid))
237                    .unwrap_or_default()
238            ),
239        }
240    }
